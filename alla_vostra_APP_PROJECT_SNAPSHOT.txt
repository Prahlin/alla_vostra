     1	import { Alert, Linking, Platform } from "react-native";
     2	
     3	export async function openPaymentLink(url) {
     4	  if (!url) {
     5	    if (Platform.OS === "web") {
     6	      window.alert("No payment link has been added for this product yet.");
     7	      return;
     8	    }
     9	
    10	    Alert.alert("Payment link missing", "No payment link has been added yet.");
    11	    return;
    12	  }
    13	
    14	  try {
    15	    const canOpen = await Linking.canOpenURL(url);
    16	
    17	    if (!canOpen) {
    18	      throw new Error("Cannot open URL");
    19	    }
    20	
    21	    await Linking.openURL(url);
    22	  } catch (error) {
    23	    if (Platform.OS === "web") {
    24	      window.alert("Unable to open this payment link.");
    25	      return;
    26	    }
    27	
    28	    Alert.alert("Unable to open link", "Please try again later.");
    29	  }
    30	}
